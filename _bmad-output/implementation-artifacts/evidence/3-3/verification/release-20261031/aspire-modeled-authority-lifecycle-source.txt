	public void Dispose()
	{
		if (_onDemandResourcesRootPath != null)
		{

				string value = await (component.Type switch
				{
					"pubsub" => GetBuildingBlockComponentAsync(component, contentWriter, "pubsub.in-memory", cancellationToken), 
					"state" => GetBuildingBlockComponentAsync(component, contentWriter, "state.in-memory", cancellationToken), 
					"secretstore" => GetBuildingBlockComponentAsync(component, contentWriter, "secretstores.local.env", cancellationToken), 
					_ => GetComponentAsync(component, contentWriter, cancellationToken), 
				}).ConfigureAwait(continueOnCapturedContext: false);
				onDemandResourcesPaths.Add(((Resource)component).Name, value);
			}
		}
		return onDemandResourcesPaths;
	}

	private async Task<string> GetComponentAsync(DaprComponentResource component, Func<string, Task<string>> contentWriter, CancellationToken cancellationToken)
	{
		logger.LogInformation("Unvalidated configuration {specType} for component '{ComponentName}'.", component.Type, ((Resource)component).Name);
		return await contentWriter(await GetDaprComponent(component, component.Type, cancellationToken)).ConfigureAwait(continueOnCapturedContext: false);
	}

	private async Task<string> GetBuildingBlockComponentAsync(DaprComponentResource component, Func<string, Task<string>> contentWriter, string defaultProvider, CancellationToken cancellationToken)
	{
		string appHostRelativePath = GetAppHostRelativePath(component.Type);
		if (File.Exists(appHostRelativePath))
		{
			logger.LogInformation("Using apphost relative path for dapr component '{ComponentName}'.", ((Resource)component).Name);
			return await contentWriter(await GetDefaultContent(component, appHostRelativePath, cancellationToken).ConfigureAwait(continueOnCapturedContext: false)).ConfigureAwait(continueOnCapturedContext: false);
		}
		string defaultComponentPath = GetDefaultComponentPath(component.Type);
		if (File.Exists(defaultComponentPath))
		{
			logger.LogInformation("Using default dapr path for component '{ComponentName}'.", ((Resource)component).Name);
			return await contentWriter(await GetDefaultContent(component, defaultComponentPath, cancellationToken).ConfigureAwait(continueOnCapturedContext: false)).ConfigureAwait(continueOnCapturedContext: false);
		}
		logger.LogInformation("Using in-memory provider for dapr component '{ComponentName}'.", ((Resource)component).Name);
		string arg = new DaprComponentSchema(((Resource)component).Name, defaultProvider).ToString();
		return await contentWriter(arg).ConfigureAwait(continueOnCapturedContext: false);
	}

	private string GetAppHostRelativePath(string componentName)
	{
		return Path.Combine(Path.Combine(GetAppHostDirectory(), ".dapr", "components"), componentName + ".yaml");
	}

	private static string GetDefaultComponentPath(string componentName)
	{
		return Path.Combine(Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.UserProfile), ".dapr", "components"), componentName + ".yaml");
	}

	private static async Task<string> GetDefaultContent(DaprComponentResource component, string defaultContentPath, CancellationToken cancellationToken)
	{
		string yamlContent = (await File.ReadAllTextAsync(defaultContentPath, cancellationToken).ConfigureAwait(continueOnCapturedContext: false)).Replace("name: " + component.Type, "name: " + ((Resource)component).Name);
		DaprComponentSchema content = DaprComponentSchema.FromYaml(yamlContent);
		await ConfigureDaprComponent(component, content, cancellationToken);
		await content.ResolveAllValuesAsync(cancellationToken);
		return content.ToString();
	}

	private static async Task<string> GetDaprComponent(DaprComponentResource component, string type, CancellationToken cancellationToken = default(CancellationToken))
	{
		DaprComponentSchema content = new DaprComponentSchema(((Resource)component).Name, type);
		await ConfigureDaprComponent(component, content, cancellationToken);
		await content.ResolveAllValuesAsync(cancellationToken);
		return content.ToString();
	}

	private static async Task ConfigureDaprComponent(DaprComponentResource component, DaprComponentSchema content, CancellationToken cancellationToken = default(CancellationToken))
	{
		IEnumerable<DaprComponentSecretAnnotation> source = default(IEnumerable<DaprComponentSecretAnnotation>);
		if (ResourceExtensions.TryGetAnnotationsOfType<DaprComponentSecretAnnotation>((IResource)(object)component, ref source) && source.Any())
		{
			content.Auth = new DaprComponentAuth
			{
				SecretStore = "secretstore"
			};
		}
		IEnumerable<DaprComponentConfigurationAnnotation> enumerable = default(IEnumerable<DaprComponentConfigurationAnnotation>);
		if (!ResourceExtensions.TryGetAnnotationsOfType<DaprComponentConfigurationAnnotation>((IResource)(object)component, ref enumerable))
		{
			return;
		}
		foreach (DaprComponentConfigurationAnnotation item in enumerable)
		{
			await item.Configure(content, cancellationToken);
		}
	}
