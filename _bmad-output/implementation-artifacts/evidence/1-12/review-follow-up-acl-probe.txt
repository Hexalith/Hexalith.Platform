package config
import "testing"
type AccessControlListOperationAction struct { OperationName string }
func TestOwnerAndRenderedParentPolicies(t *testing.T) {
  owner := NewTrie(); rendered := NewTrie()
  ownerPaths := []string{"/process", "/query", "/admin/operational-index-metadata", "/project", "/project/v2", "/project/v2/reconcile", "/replay-state", "/project/rebuild/v1", "/project/rebuild/shared/v1", "/project/rebuild/stage/v1", "/project/rebuild/commit/v1", "/project/rebuild/abort/v1", "/project/rebuild/verify/v1"}
  renderedPaths := []string{"/admin/operational-index-metadata", "/project/rebuild/commit/v1", "/project/rebuild/verify/v1", "/project/rebuild/shared/v1", "/project/rebuild/stage/v1", "/project/rebuild/abort/v1", "/project/v2/reconcile", "/project/rebuild/v1", "/replay-state", "/project/v2/*", "/project/*", "/process", "/query"}
  for _, path := range ownerPaths { owner.PutOperationAction(path, &AccessControlListOperationAction{path}) }
  for _, path := range renderedPaths { rendered.PutOperationAction(path, &AccessControlListOperationAction{path}) }
  for _, path := range []string{"/project", "/project/v2", "/project/unapproved", "/project/v2/unapproved", "/project/rebuild/verify/v1", "/project/unapproved/deeper", "/unapproved"} {
    before, after := owner.Search(path), rendered.Search(path)
    beforeName, afterName := "deny", "deny"
    if before != nil { beforeName = before.OperationName }
    if after != nil { afterName = after.OperationName }
    t.Logf("%s: owner=%s rendered=%s", path, beforeName, afterName)
    if after != nil && before == nil { t.Fatalf("rendered policy broadened %s",path) }
  }
}
