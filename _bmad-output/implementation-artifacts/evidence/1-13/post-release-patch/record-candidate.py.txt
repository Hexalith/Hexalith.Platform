def git(directory, *arguments):
    return subprocess.check_output(["git", "-C", str(directory), *arguments])

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

patches = {}
for name, directory, original, post_release, paths in (
    ("builds", builds, "5d03f1db9c38db34ecac7fe6048937478ffb82cc", "fff4a212180d8cd1c5c0759655a9bc4fb6963c45", builds_paths),
    ("platform", root, "3d51fed1f3d99acf36ef690f150136593298bd95", "258269e3896c533d763a2fdb233ed9d4287272d5", platform_paths),
):
    for label, baseline in (("original-story", original), ("post-release", post_release)):
        filename = name + "-" + label + ".patch"
        patch = git(directory, "diff", "--binary", baseline, "--", *paths)
        (evidence / filename).write_bytes(patch)
        patches[filename] = hashlib.sha256(patch).hexdigest()
